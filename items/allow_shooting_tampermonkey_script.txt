// ==UserScript==
// @name         allow shooting
// @namespace    http://tampermonkey.net
// @version      1.0
// @description  allow shooting script
// @author       MatriCYS-1
// @match        https://arras.io*
// @grant        none
// @run-at       document-start
// ==/UserScript==

(function() {
    'use strict';


    const OriginalWebSocket = window.WebSocket;


    window.WebSocket = class extends OriginalWebSocket {
        constructor(...args) {
            try {

                if (args[0] && typeof args[0] === 'string') {
                    let time = Math.round(Date.now() / 1000);


                    if (args[0].includes("/?")) {
                        args[0] = args[0].slice(0, args[0].indexOf("/?"));
                    }


                    args[0] += "/?a=3&b=8f8d16adff17e2b9&t=" + time;
                }
            } catch (e) {
                console.error("Error modifying WebSocket URL:", e);
            }


            super(...args);
        }
    };


    window.WebSocket.prototype = OriginalWebSocket.prototype;

    console.log("WebSocket connection interceptor injected successfully.");
})();
